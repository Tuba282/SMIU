public class ThrowExample {
    // Method to check age
    public static void checkAge(int age) {
        if (age < 18) { // Condition for invalid age
            throw new IllegalArgumentException("Age must be 18 or older."); // Manually throw exception
        }
        System.out.println("Access granted."); // Executes if age is valid
    }

    public static void main(String[] args) {
        checkAge(15); // Calls method with invalid age → exception thrown
    }
}
